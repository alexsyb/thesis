#Module 1: Pesticide concentration ---------------------------------------------
 
#Data import 
 compoundtab <- read.csv('MIXCOP_Compound.csv', header=TRUE, sep=';', dec='.', na.strings='-9999')
 maintab <- read.csv('MIXCOP_Main_data.csv', header=TRUE, sep=';', dec='.', na.strings='NA')
 croptab <- read.csv('MIXCOP_Crop_data.csv', header=TRUE, sep=';', dec='.', na.strings='NA')
 climate<-read.csv("")

 #Parameters:

 #number of applications
 numb=maintab[]

 #duration of simulation 
 daysofyear=maintab[]

 #description 
 description=maintab[]

 #degradation factor
 degfactor=maintab[]

 #threshold for GUTS-RED
 threshold=maintab[]

 #number of iterations
 iteration=maintab[]

 #Calculation of mixture effect: IA (Independent action) or EA (Effect addition) 
 mixture_calc=maintab[]

 #Should juvenile death-count be calculated within the model?
 juveniledeath=maintab[]
 
 
#Import crop data

 crop_name=croptab[]

 BBCH=c()
 for (i in 1:numb){
   BBCH[i]=maintab[]
 } 

 
#Import application scenario

 ar=c()
 time=c()
 
 for (i in 1:numb){
   ar[i]=maintab[] 
   time[i]=maintab[]
 }

 
#Define constant values

 #Etrans=molar enthalpy of transformation [J/mol]
 Etrans=54000

 #Tref=reference temperature 
 Tref=293.15

 #R=molar gas constant [J/mol*K]
 R=8.314

 #mixing depth soil [m]
 d=maintab[]

 #c1/c3=constants 
 c1=-0.07
 c3=100

 
#Compound information

 comid=c()
 compound=c()

 for (i in 1:numb){
   comid[i]=maintab[]
   compound[i]=as.character(compoundtab[])
 }

 
#Monthly average temperature

  pec=c()
  temp=c()
  scenario=maintab[]
  
 if (scenario==""){
   temp=c(climate[])
 }

 temp=temp+273.15

 
#CIF 

  ciftot=c()

 for (i in 1:numb){
   ciftot[i]=(croptab[])
 }

 ciftot=ciftot/100
 nsdf=1-ciftot

 
#PIEC 

 if (scenario==""){
   p=""  #soil dry bulk density
 }

 piec=c()

 depth=d*100

#The equation is based on Boesten, J. J. T. I., Helweg, A., Businelli, M., 
#Bergstrom, L., Schaefer, H., Delmas, A., ... & Yon, D. (1997). Soil persistence 
#models and EU registration. Final Report of the Work of the Soil Modelling Work 
#Group of FOCUS.

 for (i in 1:numb){
   piec[i]=((ar[i]*1000)*nsdf[i])/(100*depth*p)
 }

 
#Degradation 

 ft=c()
 degsoilt50=c()

 for (i in 1:numb){
   ft[i]=exp((-Etrans/R)*((1/Tref)-(1/temp[time[i]]))) 
   degsoilt50[i]=ft[i]*degsoil[i]
 }

 
#PEC 

 pecn=c()
 pecmatrix2 = matrix(c(0:0),nrow=daysofyear,ncol=numb)

 for(i in 1:daysofyear){
  
   dfactor=degfactor
  
   for (j in 1:numb){
     if(i==time[j]){
       pecn[j]=piec[j]
     }else{
       if (i<time[j]){
         pecn[j]=0
       }else{
         pecn[j]=piec[j]*exp(-(log(2)/(degsoilt50[j]*dfactor))*(i-time[j]))
       }
     }
   }
  
   
#Result storage

   for (j in 1:numb){
     pecmatrix2[i,j] =pecn[j]
   }
 }

 #Identical substances are combined
 comid

 colnames(pecmatrix2)<-comid

 pecnew <- by(df, INDICES=row.names(df), FUN=colSums) 
 pecnew <- as.data.frame(do.call(cbind,pecnew)) 

 write.table(pecnew,"PECsorted.csv", sep=";",dec=".", col.names=TRUE, quote=FALSE, row.names=FALSE)

 #Retrieve compound names
 newcompound <- read.csv('PECsorted.csv', header=FALSE, sep=';', dec='.', na.strings='NA')
 compoundnew<-c(newcompound[1,])
 compound2=c()

 for (i in 1:newnumb){

 }

 
#Transform into NetLogo readable Format: 

  setwd()

 pecfolcas1[1,1]=paste0("[",pecfolcas1[1,1])

 write.table(pecfolcas1,"pecfolcas1.txt",sep=" ", col.names=FALSE,row.names = FALSE, quote=FALSE)

 #Substance B:
 if (length(compound2)>1) {

 write.table(pecfolcas2,"pecfolcas2.txt",sep=" ", col.names=FALSE,row.names = FALSE, quote=FALSE)

 }
 
 #Substance C:
 if (length(compound2)>2) {
   pecfolcas3 <- matrix(c(0:0),nrow=daysofyear,ncol=100)
   
   for ( i in 1:depth[1]){
     pecfolcas3[,i]=noquote(pecnew[,3])
   }

   write.table(pecfolcas3,"pecfolcas3.txt",sep=" ", col.names=FALSE,row.names = FALSE, quote=FALSE)
   
 }

#Module 3: Link between R & NetLogo --------------------------------------------

#Open NetLogo & population model
 NLStart(nl.path, gui=TRUE, nl.jarname = "")
 model.path = Populationmodel

#Modification of NetLogo variables-----------------------------------

 NLCommand("set pesticide-application true")
 NLCommand("set toxicity-dependent-mortality true")
 NLCommand("set pesticide \"MIXCOP\"")

 #Set scenario

   NLCommand("set scenario \"")
 }

 #Set crop
if (crop_ID==""){
   NLCommand("set crop \"")
 }

 #Set mixture effect calculation
 if (mixture_calc=="IA"){
   NLCommand("set Mixture-Toxicity \"Independent action\"")
 }else{
   NLCommand("set Mixture-Toxicity \"Effect addition\"")
 }

 #Juvenile death included? 
 if (juveniledeath=="on"){
   NLCommand("set juvenile-death true")
 }else{
   NLCommand("set juvenile-death false")
 }
 
 #Number of Substances
 NLCommand("set Substances",length(compound2))

 #Set GUTS-parameter
 NLCommand("set kd1", fit1_kd)
 NLCommand("set hb1", fit1_hb)

 if (length(compound2)>1){
  
   if (threshold == "SD"){
     NLCommand("set GUTS \"GUTS-RED-SD\"")
     
     #Substance A
     NLCommand("set z1", fit1_z)
     NLCommand("set kk1", fit1_kk)
    
     #Substance B
     NLCommand("set kd2", fit2_kd)
     NLCommand("set hb2", fit2_hb)
     NLCommand("set z2", fit2_z)
     NLCommand("set kk2", fit2_kk)
    
   }else{
     NLCommand("set GUTS \"GUTS-RED-IT\"")
     #Substance A
     NLCommand("set alpha1", fit1_alpha)
     NLCommand("set beta1", fit1_beta)  
    
     #Substance B
     NLCommand("set kd2", fit2_kd)
     NLCommand("set hb2", fit2_hb)
     NLCommand("set alpha2", fit2_alpha)
     NLCommand("set beta2", fit2_beta)  
   }
  
 }else{
  
  if (threshold == "SD"){
    NLCommand("set GUTS \"GUTS-RED-SD\"")
    NLCommand("set z1", fit1_z)
    NLCommand("set kk1", fit1_kk)
    }else{
    NLCommand("set GUTS \"GUTS-RED-IT\"")
    NLCommand("set alpha1", fit1_alpha)
    NLCommand("set beta1", fit1_beta)  
  }
 }

 #Matrices for iteration

 minage=c()

 meandepth=c()
 lowerdepth=c()
 upperdepth=c()


#Setup & run Populationmodel
 NLCommand("setup") 

 for (i in 1: daysofyear){

   NLDoCommand(1,"go")
  
   countadults[i]<-NLReport("count adults")
   countjuveniles[i]<-NLReport("count juveniles")
   counteggs[i]<-NLReport("count eggs")

  if (countadults[i]>0) {
   meanconcseeneachstep1[i]<-NLReport("mean [conc-seen-total-mean1] of adults")
   meanconcseeneachstep2[i]<-NLReport("mean [conc-seen-total-mean2] of adults")
   dailytoxdeath[i]<-NLReport("daily-toxdeathcount")
   meandeathrate[i]<-NLReport("mean [Totaldeath] of adults")
   maxdeathrate[i]<-NLReport("max [Totaldeath] of adults")
   
   meanlifespan[i]<-NLReport("mean [life-span] of adults")
   meanage[i]<-NLReport("mean [age] of adults")
   maxage[i]<-NLReport("max [age] of adults")
   minage[i]<-NLReport("min [age] of adults")
  
   if (i>1){
    if (countadults[i-1]==0){
      dailytoxdeathpercentage[i]=0
    }else{
      dailytoxdeathpercentage[i]=((dailytoxdeath[i]/countadults[i-1])*100)
    }
   }
  }else{
    meanconcseeneachstep1[i] = 0
    meanconcseeneachstep2[i] = 0
    dailytoxdeath[i] = 0
    meandeathrate[i] = 0
    maxdeathrate[i] = 0
    dailytoxdeathpercentage[i] = 0
  }
   
  meandepth[i]=NLReport("meandepth")
  lowerdepth[i]=NLReport("lowerdepth")
  upperdepth[i]=NLReport("upperdepth")
  
  countadultsall[z,i]=countadults[i]
  countjuvenilesall[z,i]=countjuveniles[i]

  meanconcseeneachstepall1[z,i]=meanconcseeneachstep1[i]
  meanconcseeneachstepall2[z,i]=meanconcseeneachstep2[i]
  dailytoxdeathall[z,i]=dailytoxdeath[i]
  dailytoxdeathpercentageall[z,i]=dailytoxdeathpercentage[i]
  meantotaldeath[z,i]=meandeathrate[i]
  maxtotaldeath[z,i]=maxdeathrate[i]
  
  meandepthall[z,i]=meandepth[i]
  lowerdepthall[z,i]=lowerdepth[i]
  upperdepthall[z,i]=upperdepth[i]
  
  ageall[z,i]=meanage[i]

}
