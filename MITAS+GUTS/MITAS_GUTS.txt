#duration of simulation in days

description=maintab[1,'description']

   BBCH[i]=maintab[i,'BBCH']

   ar[i]=maintab[i,''] 
   time[i]=maintab[i,''] 
}

#Modul 1.5: define constant values----------------------------------------
#Etrans=molar enthalpy of transformation [J/mol]
Etrans=54000

#Tref=reference temperature 
Tref=293.15

#R=molar gas constant [J/mol*K]
R=8.314

#mixing depth soil [m]
d=maintab[1,'']

#c1/c3=constants 
c1=-0.07
c3=100

#Modul 1.7: Missing ------------------------------------------------------

chemcla=c()
chemuse=c()

lc50earth=c()  
noecearth=c() 
lc50folsomia=c() 

for (i in 1:numb){
   chemcla[i]=compoundtab[comid[i],'']
   chemuse[i]=compoundtab[comid[i],'']
   degsoil[i]=compoundtab[comid[i],''] 
   
   lc50earth[i]=compoundtab[comid[i],'']
   noecearth[i]=compoundtab[comid[i],'']
   lc50folsomia[i]=compoundtab[comid[i],'']
}

comchemclas=c(compoundtab$Chemical.class)
comchemuse=c(compoundtab$Chemical.use)
comT50soil=c(compoundtab$DT50soil..d.)
commatrix=rbind(comchemclas,comchemuse,comT50soil)

alllc50earth=c(compoundtab$LC50.earthworm..mg.ai.kg.dw.)
allnoecearth=c(compoundtab$NOEC.earthworm..mg.ai.kg.dw.)

same=c()
averagelc50=c()
averagenoec=c()


#loop for DegT50soilref:
for (i in 1:numb){
   if (is.na(degsoil[i])){
      same=commatrix[3,commatrix[1,]==chemcla[i]] 
      degsoil[i]=mean(same,na.rm=TRUE)
   }
   if (is.na(degsoil[i])){
      same=commatrix[3,commatrix[2,]==chemuse[i]]
      degsoil[i]=mean(same,na.rm=TRUE)
   }
}

#loop for LC50:
for (i in 1:numb){
  if (is.na(lc50earth[i])){
    averagelc50=commatrix[4,commatrix[1,]==chemcla[i]] 
    lc50earth[i]=mean(averagelc50,na.rm=TRUE)
  }
  if (is.na(lc50earth[i])){
    averagelc50=commatrix[4,commatrix[2,]==chemuse[i]]
    lc50earth[i]=mean(averagelc50,na.rm=TRUE)
  }
}

#loop for NOEC:
for (i in 1:numb){
  if (is.na(noecearth[i])){
    averagenoec=commatrix[5,commatrix[1,]==chemcla[i]] 
    noecearth[i]=mean(averagenoec,na.rm=TRUE)
  }
  if (is.na(noecearth[i])){
    averagenoec=commatrix[5,commatrix[2,]==chemuse[i]]
    noecearth[i]=mean(averagenoec,na.rm=TRUE)
  }
}

#Modul 1.8: determination of the vectors----------------------------------

pec=c()
etr=c()
teracute=c()
terchronic=c()
toxchron=c()
toxchronic=c()
etrfolsomia=c()
group=c()

#Modul 1.9: monthly average temperature-----------------------------------

temp=c()
scenario=maintab[1,'']
if (scenario==''){
   temp=c(climate[ ,1])
}

temp=temp+273.15

#Modul 1.10: determine CIF-value------------------------------------------

ciftot=c()

for (i in 1:numb){
   ciftot[i]=(croptab[crop_ID,(BBCH[i]+2)])
}

ciftot=ciftot/100

#-----NSDF(Net Soil Deposition Factor)-------------------
nsdf=1-ciftot

#Modul 1.11: PIEC (Predicted Initial Environmental concentration)---------


if (scenario==''){
   p=1.3
}

piec=c()

depth=d*100

#The equation is based on Boesten, J. J. T. I., Helweg, A., Businelli, M., 
#Bergstrom, L., Schaefer, H., Delmas, A., ... & Yon, D. (1997). Soil persistence 
#models and EU registration. Final Report of the Work of the Soil Modelling Work 
#Group of FOCUS.
for (i in 1:numb){
   piec[i]=((ar[i]*1000)*nsdf[i])/(100*depth*p)
}

#Modul 1.12: disappearence rate constant----------------------------------

ft=c()
degsoilt50=c()


for (i in 1:numb){
  #temperature correction factor based on the Arrhenius eqation
   ft[i]=exp((-Etrans/R)*((1/Tref)-(1/temp[time[i]]))) 
   degsoilt50[i]=ft[i]*degsoil[i]
}

#Modul 1.13: PEC and acute/chronic ETR------------------------------------

pecn=c()
etrn=c()
etrf=c()
tern=c()
terch=c()
toxchron=c()
etria=c()
pecmatrix = matrix(c(0:0),nrow=numb,ncol=daysofyear)
pecmatrix2 = matrix(c(0:0),nrow=daysofyear,ncol=numb)
etrmatrix = matrix(c(0:0),nrow=numb,ncol=daysofyear)
etrmatrix2 = matrix(c(0:0),nrow=daysofyear,ncol=numb)
toxchronmatrix=matrix(c(0:0),nrow=daysofyear,ncol=numb)

for(i in 1:daysofyear){
   
   dfactor=degfactor
   
   for (j in 1:numb){
      if(i==time[j]){
         pecn[j]=piec[j]
      }else{
         if (i<time[j]){
            pecn[j]=0
         }else{
            pecn[j]=piec[j]*exp(-(log(2)/(degsoilt50[j]*dfactor))*(i-time[j]))
         }
      }
   }
   
   
   for (j in 1:numb){
     etrn[j]=pecn[j]/(lc50earth[j])
     toxchron[j]=pecn[j]/(noecearth[j])
   }
   
   #Modul 1.14: result storage-----------------------------------------------

   for (j in 1:numb){
      pecmatrix[j,i] =pecn[j] 
      pecmatrix2[i,j] =pecn[j] 
      etrmatrix[j,i] =etrn[j]
      etrmatrix2[i,j] =etrn[j]
      toxchronmatrix[i,j] =toxchron[j]
   }
   
   #total toxicities (sums):
   pec[i]=sum(pecn) 
   etr[i]=sum(etrn)
   toxchronic[i]=sum(toxchron)
}

#et al. (2013) Concentration Addition, Independent Action and Generalized Concentration 
#Addition Models for Mixture Effect Prediction of Sex Hormone Synthesis In Vitro.

     ca[i]= sum(allmortalitymatrix[i,])

   }
   
   if (newnumb>2){
     ia[i]=1-((1-allmortalitymatrix[i,1])*(1-allmortalitymatrix[i,2])*(1-allmortalitymatrix[i,3]))
     ca[i]= sum(allmortalitymatrix[i,])

   }
   
 }
